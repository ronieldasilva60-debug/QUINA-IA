<script>

function gerar() {

  setStatus("Gerando Base IA...");

  google.script.run.withSuccessHandler(render).atualizarQuinaIA();

}

function atualizar() {

  setStatus("Importando últimos 200 concursos...");

  google.script.run
    .withSuccessHandler((msg) => {

      setStatus(msg);

      carregarRanking();
      carregarBase();
      carregarJogos();

    })
    .importarUltimosConcursos();

}

function render() {

  setStatus("Atualizado com sucesso ✔");

  carregarBase();
  carregarRanking();
  carregarJogos();

}

function carregarBase() {

  google.script.run.withSuccessHandler(data => {

    const div = document.getElementById("base");

    div.innerHTML = "";

    data.forEach(n => {

      const el = document.createElement("div");

      el.className = "ball";

      el.innerText = n;

      div.appendChild(el);

    });

  }).getBase();

}

function carregarRanking() {

  google.script.run.withSuccessHandler(data => {

    let html = "<tr><th>Número</th><th>Score</th></tr>";

    data.slice(0,10).forEach(r => {

      html += `<tr>
        <td>${r.number}</td>
        <td>${r.score.toFixed(2)}</td>
      </tr>`;

    });

    document.getElementById("ranking").innerHTML = html;

  }).getRanking();

}

function carregarJogos() {

  google.script.run.withSuccessHandler(data => {

    let html = "<tr><th>Jogo</th><th>Dezenas</th></tr>";

    data.forEach((j,i) => {

      html += `<tr>
        <td>${i+1}</td>
        <td>${j.join(" - ")}</td>
      </tr>`;

    });

    document.getElementById("jogos").innerHTML = html;

  }).getJogos();

}

function conferirUltimaGeracao() {

  google.script.run.withSuccessHandler(function(data) {

    document.getElementById("resultado-quina").innerHTML =
      "<b>Último resultado:</b> " + data.resultado.join(" - ");

    let html = "";

    data.jogos.forEach(j => {
      html += `
        <div style="margin-top:10px;">
          Jogo ${j.index}: ${j.jogo.join(" - ")}
          <b style="color:gold;"> | Acertos: ${j.acertos}</b>
        </div>
      `;
    });

    document.getElementById("lista-jogos").innerHTML = html;

  }).conferirUltimaGeracao();
}

function setStatus(msg) {

  document.getElementById("status").innerText = msg;

}

window.onload = function() {
  carregarBase();
  carregarRanking();
  carregarJogos();
}
