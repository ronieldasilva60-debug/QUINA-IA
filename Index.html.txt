<script>
function gerar() {
  setStatus("Gerando 30 bases candidatas e jogos distribuídos...");
  const tipo = document.getElementById("tipoBase").value;
  google.script.run.withSuccessHandler(function() {
    render();
  }).withFailureHandler(function(err) {
    setStatus("Erro: " + err.message);
  }).atualizarQuinaIA(tipo);
}

function atualizar() {
  setStatus("Importando últimos 200 concursos...");
  google.script.run.withSuccessHandler(function(msg) {
    setStatus(msg);
    carregarRanking(); carregarBase(); carregarJogos(); carregarMultiBase();
  }).withFailureHandler(function(err) {
    setStatus("Erro: " + err.message);
  }).importarUltimosConcursos();
}

function render() {
  setStatus("Atualizado com sucesso ✔");
  carregarBase(); carregarRanking(); carregarJogos(); carregarMultiBase();
}

function carregarBase() {
  google.script.run.withSuccessHandler(function(data) {
    const div = document.getElementById("base");
    div.innerHTML = "";
    data.forEach(function(n) {
      const el = document.createElement("div");
      el.className = "ball";
      el.innerText = String(n).padStart(2, "0");
      div.appendChild(el);
    });
  }).getBase();
}

function carregarMultiBase() {
  google.script.run.withSuccessHandler(function(data) {
    document.getElementById("bases-analisadas").innerText = data.basesAnalisadas || 0;
    const div = document.getElementById("multi-base");
    div.innerHTML = "";
    (data.topBases || []).forEach(function(base) {
      const card = document.createElement("div");
      card.className = "base-card";
      card.innerHTML = `
        <h4>${base.titulo}</h4>
        <div class="metric">Estratégia: <b>${base.estrategia}</b></div>
        <div class="metric">Score: <b>${Number(base.score).toFixed(1)}</b></div>
        <div class="metric">Jogos gerados: <b>${base.qtdJogos || 0}</b></div>
        <div class="numbers">${base.dezenas.map(n => `<span class="ball">${String(n).padStart(2, "0")}</span>`).join("")}</div>
      `;
      div.appendChild(card);
    });
  }).getMultiBaseResumoQuina();
}

function carregarRanking() {
  google.script.run.withSuccessHandler(function(data) {
    let html = "<tr><th>Número</th><th>Score</th></tr>";
    data.slice(0, 10).forEach(function(r) {
      html += `<tr><td>${String(r.number).padStart(2, "0")}</td><td>${r.score.toFixed(2)}</td></tr>`;
    });
    document.getElementById("ranking").innerHTML = html;
  }).getRanking();
}

function carregarJogos() {
  google.script.run.withSuccessHandler(function(data) {
    let html = "<tr><th>Jogo</th><th>Dezenas</th></tr>";
    data.forEach(function(j, i) {
      html += `<tr><td>${i + 1}</td><td>${j.map(n => String(n).padStart(2, "0")).join(" - ")}</td></tr>`;
    });
    document.getElementById("jogos").innerHTML = html;
  }).getJogos();
}

function conferirUltimaGeracao() {
  google.script.run.withSuccessHandler(function(data) {
    document.getElementById("resultado-quina").innerHTML = "<b>Último resultado:</b> " + data.resultado.map(n => String(n).padStart(2, "0")).join(" - ");
    let html = "";
    data.jogos.forEach(function(j) {
      html += `<div style="margin-top:10px;">Jogo ${j.index}: ${j.jogo.map(n => String(n).padStart(2, "0")).join(" - ")} <b style="color:gold;"> | Acertos: ${j.acertos}</b></div>`;
    });
    document.getElementById("lista-jogos").innerHTML = html;
  }).conferirUltimaGeracao();
}

function setStatus(msg) { document.getElementById("status").innerText = msg; }

window.onload = function() { carregarBase(); carregarRanking(); carregarJogos(); carregarMultiBase(); };
</script>
